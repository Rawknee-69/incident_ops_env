Quick Start

import asyncio
from incident_ops_env import IncidentAction, IncidentOpsEnv

async def main():
    async with IncidentOpsEnv(base_url="http://localhost:7860") as env:
        result = await env.reset(task_id=1, seed=42)
        print("Alerts:", len(result.observation.active_alerts))

        action = IncidentAction(
            action_type="classify_alert",
            severity="P2",
            service_name="payment-service",
            pattern_type="database_overload",
        )
        result = await env.step(action)
        print("Reward:", result.reward, "Done:", result.done)

asyncio.run(main())
import os
from openai import OpenAI
import httpx

client = OpenAI()
ENV_URL = "http://localhost:7860"

# Reset environment
reset = httpx.post(f"{ENV_URL}/reset", json={"task_id": 1, "seed": 42}).json()
session_id = reset["session_id"]
observation = reset["observation"]

# Build prompt from observation
prompt = f"""You are an SRE on-call. Classify this alert.
Alerts: {observation['active_alerts']}
Respond with JSON: {{"severity": "P1|P2|P3", "service_name": "...", "pattern_type": "..."}}"""

response = client.chat.completions.create(
    model="gpt-4o-mini",
    messages=[{"role": "user", "content": prompt}],
    response_format={"type": "json_object"}
)

# Parse and send action
import json
action_data = json.loads(response.choices[0].message.content)
action_data["action_type"] = "classify_alert"

result = httpx.post(
    f"{ENV_URL}/step",
    json={"action": action_data},
    headers={"X-Session-ID": session_id}
).json()
print("Reward:", result["reward"], "Done:", result["done"])
import os
import google.generativeai as genai
import httpx
import json

genai.configure(api_key=os.environ["GEMINI_API_KEY"])
model = genai.GenerativeModel("gemini-2.0-flash")
ENV_URL = "http://localhost:7860"

# Reset environment
reset = httpx.post(f"{ENV_URL}/reset", json={"task_id": 1, "seed": 42}).json()
session_id = reset["session_id"]
observation = reset["observation"]

# Build prompt
prompt = f"""You are an SRE on-call. Classify this alert.
Alerts: {observation['active_alerts']}
Respond with JSON only: {{"severity": "P1|P2|P3", "service_name": "...", "pattern_type": "..."}}"""

response = model.generate_content(prompt)

# Parse and send action
action_data = json.loads(response.text)
action_data["action_type"] = "classify_alert"

result = httpx.post(
    f"{ENV_URL}/step",
    json={"action": action_data},
    headers={"X-Session-ID": session_id}
).json()
print("Reward:", result["reward"], "Done:", result["done"])
